# mlx-vlm ships the OpenAI server the MLX engine spawns. It unlocks native tool
# calling (no mlx_lm.server EOS-flush bug) and image input. Pulls Pillow/opencv/
# mlx-audio/python-multipart transitively (the inference path keeps cv2/mlx_audio
# lazy).
# 0.6.13 (#273): datasets moved out of the core deps (train extra only), mlx
# floor is now >=0.32.0, and transformers must be >=5.14.0 (pinned in base.txt).
# 0.6.17 (#378): the Qwen2.5-VL / Qwen2-VL vision towers cast the mx.repeat
# count to int; 0.6.13-0.6.16 pass an mx.array, which mlx>=0.32.2 rejects with
# a TypeError on every image. 0.6.17 also drops mlx-lm from its dependencies
# (text-only architectures are native modules now), so mlx-lm is no longer
# installed on a fresh setup and nothing here imports it.
mlx-vlm==0.6.17

# mlx is pinned explicitly since #378: unpinned, the 2.0.0 release build
# resolved 0.32.2 (strict `repeat(array, repeats: int)` signature) while dev
# venvs sat on older wheels that still accepted an array, so the packaged app
# was the only place the broken combination existed. Keep it on the exact
# version the release build resolves so dev reproduces prod.
mlx==0.32.2

# transformers is pinned in meta/base.txt (5.14.1, normalized for all platforms).
# Its floor is dictated here by mlx-vlm 0.6.17 (requires transformers>=5.14.0).
# mac resolves torch (>=2.7, has float8_e8m0fnu) transitively via the mlx/ST stack.
